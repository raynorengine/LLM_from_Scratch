# 우리가 채운다(= 이 말로 시작해서 이어 봐). 프롬프트도 문장의 일부다.
#

# bos_left	0	(문맥에 남은 BOS. 0이면 문장 첫머리임을 모른다)
# candidates	0	(이 문맥의 다음 토큰 후보 수. 0이면 막힘)
# context_total	0
# corpus_starts	0	(원본에서 이 프롬프트로 시작하는 문장 수)
#
# seed	1792779963
# count	15
# n	6
# vocab	16920
# sentences	64376
# tokens	1460593
# total_grams	1524969
# contexts	826271
# build_ms	337.000
# gen_ms	0.000
# avg_tokens	5.00
# hit_limit	0
# blocked	15	(첫 문맥에 후보가 없어 생성하지 못한 횟수)

## 막혔다 - 이 프롬프트로는 생성할 수 없다
#
# 위 context 가 학습 데이터(64376문장)에 한 번도 나오지 않는다.
